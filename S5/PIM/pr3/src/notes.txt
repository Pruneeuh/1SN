   function Codage_8_bits(Nb : in Integer) return Unbounded_String is 
         Stock : Integer; 
         Bit : Integer; 
         Octet : Unbounded_String; 
   begin 
         Stock := Nb;
         for indice in 1..8 loop 
            Bit := Stock mod 2; 
            Octet := Octet & To_Unbounded_String(Bit); 
         end loop; 
         return Octet;
   end Codage_8_bits;
